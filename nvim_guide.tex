% --- General & Core UX ---
\section{General \& Core UX}
\begin{tabular}{@{}ll@{}}
\texttt{<Leader>} & Space key \\
\texttt{<Esc>} & Clear search highlights \\
\texttt{<leader>q} & Open diagnostic [Q]uickfix list \\
\texttt{<leader>f} & [F]ormat buffer (\texttt{conform.nvim}) \\
\texttt{<leader>z} & Toggle [Z]en Mode (\texttt{zen-mode}) \\
\texttt{gcc} / \texttt{gc} & Comment line (Normal / Visual) \\
\texttt{gbc} / \texttt{gb} & Comment block (Normal / Visual) \\
\texttt{u} / \texttt{CTRL-r} & [u]ndo / [r]edo change \\
\texttt{.} & Repeat last edit or command \\
\texttt{<Esc><Esc>} & Exit builtin terminal mode \\
\end{tabular}

% --- Buffer & File Management ---
\section{File Explorer \& Buffers}
\begin{tabular}{@{}ll@{}}
\texttt{-} & Open [Oil.nvim] parent directory \\
\texttt{\_} & [Oil]: Open working directory (CWD) \\
\texttt{<BS>} / \texttt{h} & [Oil]: Go up one directory \\
\texttt{l} & [Oil]: Open file / enter folder \\
\texttt{g.} & [Oil]: Toggle hidden files \\
\texttt{<leader>bd} & Smart [B]uffer [D]elete (returns to Oil) \\
\texttt{:bd} / \texttt{:bd!} & Aliased to smart buffer delete \\
\texttt{<C-h,j,k,l>} & Move focus to Window (Left/Down/Up/Right) \\
\end{tabular}

% --- Vim Text Motions ---
\section{Vim Text Motions}
\begin{tabular}{@{}ll@{}}
\texttt{h, j, k, l} & Move Left, Down, Up, Right \\
\texttt{w} / \texttt{b} & Move [w]ord forward / [b]ackward \\
\texttt{e} / \texttt{ge} & End of word / End of previous word \\
\texttt{0} / \texttt{\$} & Beginning / End of line \\
\texttt{\^{}} / \texttt{g\_} & First / Last non-blank character \\
\texttt{\{} / \texttt{\}} & Previous / Next paragraph \\
\texttt{\%} & Jump to matching pair \texttt{()[]\{\}} \\
\texttt{gg} / \texttt{G} & Top / Bottom of buffer \\
\texttt{M, H, L} & [M]iddle, [H]igh, [L]ow of screen \\
\texttt{CTRL-u / d} & Scroll half-page [u]p / [d]own \\
\texttt{f/t <char>} & Find / Till next char on line \\
\texttt{F/T <char>} & Find / Till previous char on line \\
\texttt{; / ,} & Repeat last find/till (fwd / rev) \\
\texttt{*} / \texttt{\#} & Search word under cursor (fwd / rev) \\
\end{tabular}
